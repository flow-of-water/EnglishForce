### HTTP

server {
    listen 80;
    listen [::]:80;
    server_name 20.187.151.69 englishforce.site www.englishforce.site;

    root /home/azureuser/apps/EnglishForce/EnglishForce-frontend/build;
    index index.html;

    location / {
        try_files $uri /index.html;
    }

    location /api {
        proxy_pass http://localhost:5000;  # <== backend bạn đang dùng PORT=5000
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}

server {
    server_name 20.187.151.69 englishforce.site www.englishforce.site;

    root /home/azureuser/apps/EnglishForce/EnglishForce-frontend/build;
    index index.html;

    location / {
        try_files $uri /index.html;
    }

    location /api {
        proxy_pass http://localhost:5000;  # <== backend bạn đang dùng PORT=5000
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/englishforce.site/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/englishforce.site/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}

server {
    if ($host = www.englishforce.site) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    if ($host = englishforce.site) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    listen [::]:80;
    server_name 20.187.151.69 englishforce.site www.englishforce.site;
    return 404; # managed by Certbot
}
